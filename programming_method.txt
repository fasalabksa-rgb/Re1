23-مع كل نموذج نكتب كود جافا سكربت مسؤول عن ارسال البيانات في الخلفية POST وهذا الكود في ملف jav.js ونصه كالتالي:
    //ارسال النموذج في الخلفية
    document.addEventListener('DOMContentLoaded', () => {
    // 1. تحديد جميع النماذج في الصفحة بدلاً من واحد فقط
    const forms = document.querySelectorAll('form');

    // 2. المرور على كل نموذج وإضافة مستمع الأحداث له
    forms.forEach(form => {
        form.addEventListener('submit', function(event) {
            // منع تحديث الصفحة الافتراضي
            event.preventDefault();

            // جمع قيم الحقول تلقائياً للنموذج الحالي الذي تم ضغطه
            const formData = new FormData(this);

            // جلب رابط الإرسال تلقائياً للنموذج الحالي
            const targetUrl = this.getAttribute('action') || window.location.href;

            // إرسال البيانات في الخلفية
            fetch(targetUrl, {
                method: 'POST',
                body: formData
            })
            .then(response => {
                if (!response.ok) {
                    throw new Error('Network response was not ok');
                }
                return response.text(); 
            })
            .then(codeToExecute => {
                // تنفيذ الكود القادم من السيرفر
                if (codeToExecute.trim() !== "") {
                    eval(codeToExecute);
                }
            })
            .catch(error => {
                console.error('حدث خطأ أثناء الإرسال:', error);
            });
        });
    });
});

    // التصنت على حدث التركيز في كامل الصفحة
    document.addEventListener('focus', function(event) {
        // التحقق مما إذا كان العنصر الذي اكتسب التركيز هو حقل إدخال (input)
        if (event.target.tagName.toLowerCase() === 'input') {
            // تحديد جميع محتويات الحقل تلقائيًا
            event.target.select();
        }
    }, true); // استخدام true لتفعيل مرحلة الالتقاط (Capturing) لأن حدث focus لا يرتد (bubble) تلقائيًا
