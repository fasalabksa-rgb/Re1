9-اي فنكشن جديد تضعه في ملف function.php

18-إن القالب الرئيسي للموقع الهيدر والفوتر هو layout.html والمتغير الذي يعبر عن محتوى الصفحة $body
19-عندما تبرمج بلغة جافا سكربت ابتعد قدر الإمكان addEventListener وضع لكل وسم الحدث والداله الخاصه بها لكي تسهل المراجعه والتعديل
20-يتنقل المستخدم بين الحقول بسهم يمين ويسار عندما يكون المؤشر في بداية ونهاية الحقل النصي (<input type="text") ويدخل المستخدم البيانات ويعتمدها بالضغط على enter وعند الضغط على enter يُدخل البيانات ويعتمدها ويذهب التركيز للحقل التالي
21-يجب أن لا تتذكر الحقول المدخلات السابقة دائما.
22-اسماء حقول النموذج يجب ان تكون مطابقة لأسماء حقول جداول قواعد البيانات.
23-مع كل نموذج نكتب كود جافا سكربت مسؤول عن ارسال البيانات في الخلفية POST وهذا الكود في ملف jav.js ونصه كالتالي:
    //ارسال النموذج في الخلفية
    document.addEventListener('DOMContentLoaded', () => {
    // 1. تحديد جميع النماذج في الصفحة بدلاً من واحد فقط
    const forms = document.querySelectorAll('form');

    // 2. المرور على كل نموذج وإضافة مستمع الأحداث له
    forms.forEach(form => {
        form.addEventListener('submit', function(event) {
            // منع تحديث الصفحة الافتراضي
            event.preventDefault();

            // جمع قيم الحقول تلقائياً للنموذج الحالي الذي تم ضغطه
            const formData = new FormData(this);

            // جلب رابط الإرسال تلقائياً للنموذج الحالي
            const targetUrl = this.getAttribute('action') || window.location.href;

            // إرسال البيانات في الخلفية
            fetch(targetUrl, {
                method: 'POST',
                body: formData
            })
            .then(response => {
                if (!response.ok) {
                    throw new Error('Network response was not ok');
                }
                return response.text(); 
            })
            .then(codeToExecute => {
                // تنفيذ الكود القادم من السيرفر
                if (codeToExecute.trim() !== "") {
                    eval(codeToExecute);
                }
            })
            .catch(error => {
                console.error('حدث خطأ أثناء الإرسال:', error);
            });
        });
    });
});
